var SH_EXAM = 'DeThi', SH_ATT = 'LuotThi', SH_RES = 'KetQua';
var EXAM_HEAD = ['Mã đề', 'Tiêu đề', 'Loại bài', 'Môn học/Mô đun', 'Lớp', 'Trạng thái', 'Tạo lúc', 'Cập nhật', 'Số phần JSON'];
var ATT_HEAD = ['Mã lượt', 'Mã đề', 'Seed', 'Họ và tên', 'Lớp', 'MSSV', 'Bắt đầu', 'Trạng thái', 'Nộp lúc', 'Khóa', 'Câu hiện tại', 'Bắt đầu câu', 'Tạm lưu', 'Vi phạm', 'Lần'];
var RES_HEAD = ['Mã lượt', 'Mã đề', 'Họ và tên', 'Lớp', 'MSSV', 'Điểm', 'Số câu đúng', 'Số câu TN', 'Điểm TN đạt', 'Điểm TL đạt', 'Tổng điểm tối đa', 'Tự luận', 'Bắt đầu', 'Nộp bài', 'Thời gian làm (giây)', 'Vi phạm', 'Ghi chú', 'Chi tiết', 'Điểm TL', 'Lần', 'Kỹ năng'];
var CHUNK = 45000;

function doGet(e) {
  return out_({ ok: true, app: 'QuizApp PV College', version: QC_VERSION, time: new Date().toISOString() });
}

function doPost(e) {
  var req;
  try { req = JSON.parse(e.postData.contents); } catch (err) { return out_({ ok: false, error: 'Dữ liệu gửi lên không hợp lệ' }); }
  try {
    var r = route_(req || {});
    r.ok = r.ok !== false;
    return out_(r);
  } catch (err) {
    return out_({ ok: false, error: String(err && err.message || err) });
  }
}

function out_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

function teacherKey_() {
  var p = PropertiesService.getScriptProperties().getProperty('TEACHER_KEY');
  return p || TEACHER_KEY;
}
function needTeacher_(req) { if (String(req.key || '') !== String(teacherKey_())) throw new Error('Sai mật khẩu giáo viên'); }

function route_(req) {
  switch (req.action) {
    case 'ping': return { version: QC_VERSION, teacher: req.key ? String(req.key) === String(teacherKey_()) : undefined };
    // ---- Giáo viên ----
    case 'saveExam': needTeacher_(req); return saveExam_(req.exam);
    case 'listExams': needTeacher_(req); return { exams: listExams_() };
    case 'getExam': needTeacher_(req); return { exam: loadExam_(req.examId) };
    case 'setStatus': needTeacher_(req); return setStatus_(req.examId, req.status);
    case 'deleteExam': needTeacher_(req); return deleteExam_(req.examId);
    case 'results': needTeacher_(req); return results_(req.examId);
    case 'gradeEssay': needTeacher_(req); return gradeEssay_(req.attemptId, req.scores);
    case 'deleteAttempt': needTeacher_(req); return deleteAttempt_(req.attemptId);
    // ---- Học sinh ----
    case 'info': return { info: QC_publicInfo(openExam_(req.examId)), serverNow: Date.now() };
    case 'start': return start_(req);
    case 'progress': return progress_(req);
    case 'submit': return submit_(req);
    case 'audio': return audio_(req);
    case 'uploadSpeech': return uploadSpeech_(req);
    // ---- Tệp âm thanh (đề Tiếng Anh) ----
    case 'uploadAudio': needTeacher_(req); return uploadAudio_(req);
    case 'speech': needTeacher_(req); return speech_(req);
  }
  throw new Error('Yêu cầu không hợp lệ: ' + req.action);
}

/* ------------------------------ Sheet helpers ------------------------------ */
var TEXT_COLS = { DeThi: [1, 2, 3, 4, 5], LuotThi: [1, 2, 3, 4, 5, 6, 10, 13], KetQua: [1, 2, 3, 4, 5, 12, 17, 18, 19] };
function sheet_(name, head) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    (TEXT_COLS[name] || []).forEach(function (c) { sh.getRange(1, c, sh.getMaxRows(), 1).setNumberFormat('@'); });
    sh.getRange(1, 1, 1, head.length).setValues([head]).setFontWeight('bold').setBackground('#0b3d5c').setFontColor('#ffffff');
    sh.setFrozenRows(1);
  } else if (sh.getLastColumn() < head.length) {
    sh.getRange(1, 1, 1, head.length).setValues([head]).setFontWeight('bold').setBackground('#0b3d5c').setFontColor('#ffffff');
  }
  return sh;
}
function rows_(sh) {
  var n = sh.getLastRow();
  if (n < 2) return [];
  return sh.getRange(2, 1, n - 1, sh.getLastColumn()).getValues();
}
function findRow_(sh, col, val) {
  var n = sh.getLastRow();
  if (n < 2) return -1;
  var v = sh.getRange(2, col, n - 1, 1).getValues();
  for (var i = 0; i < v.length; i++) if (String(v[i][0]) === String(val)) return i + 2;
  return -1;
}
function withLock_(fn) {
  var lock = LockService.getScriptLock();
  lock.waitLock(25000);
  try { return fn(); } finally { lock.releaseLock(); }
}
function newId_(len) {
  var s = '', c = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  for (var i = 0; i < (len || 6); i++) s += c.charAt(Math.floor(Math.random() * c.length));
  return s;
}

/* --------------------------------- Đề thi --------------------------------- */
function saveExam_(exam) {
  if (!exam || !exam.banks || !exam.groups) throw new Error('Đề trống');
  return withLock_(function () {
    var sh = sheet_(SH_EXAM, EXAM_HEAD);
    if (!exam.id) { do { exam.id = newId_(6); } while (findRow_(sh, 1, exam.id) > 0); }
    var json = JSON.stringify(exam);
    var parts = [];
    for (var i = 0; i < json.length; i += CHUNK) parts.push('~' + json.substring(i, i + CHUNK));
    var info = exam.info || {};
    var row = findRow_(sh, 1, exam.id);
    var now = new Date();
    var status = exam.status || 'Mở';
    if (row < 0) { row = sh.getLastRow() + 1; sh.getRange(row, 7).setValue(now); }
    else { var oldParts = +sh.getRange(row, 9).getValue() || 0; if (oldParts > parts.length) sh.getRange(row, 10 + parts.length, 1, oldParts - parts.length).clearContent(); }
    sh.getRange(row, 1, 1, 6).setValues([[exam.id, exam.title || '', info.loai || '', info.mon || '', info.lop || '', status]]);
    sh.getRange(row, 8, 1, 2).setValues([[now, parts.length]]);
    sh.getRange(row, 10, 1, parts.length).setNumberFormat('@').setValues([parts]);
    return { examId: exam.id };
  });
}
function loadExam_(id) {
  var sh = sheet_(SH_EXAM, EXAM_HEAD);
  var row = findRow_(sh, 1, id);
  if (row < 0) throw new Error('Không tìm thấy đề có mã ' + id);
  var n = +sh.getRange(row, 9).getValue();
  var parts = sh.getRange(row, 10, 1, n).getValues()[0];
  var exam = JSON.parse(parts.map(function (p) { return String(p).replace(/^~/, ''); }).join(''));
  exam.status = sh.getRange(row, 6).getValue();
  return exam;
}
function openExam_(id) {
  var exam = loadExam_(id);
  if (String(exam.status) !== 'Mở') throw new Error('Đề thi này hiện đang ĐÓNG. Vui lòng liên hệ giáo viên.');
  return exam;
}
/** Giờ Việt Nam dd/mm/yyyy hh:mm */
function fmtVN_(ts) {
  var d = new Date(+ts + 7 * 3600000), p = function (n) { return (n < 10 ? '0' : '') + n; };
  return p(d.getUTCHours()) + ':' + p(d.getUTCMinutes()) + ' ngày ' + p(d.getUTCDate()) + '/' + p(d.getUTCMonth() + 1) + '/' + d.getUTCFullYear();
}
/** Kiểm tra khung giờ: chưa mở / hết giờ vào làm (trừ khi đang làm dở) / đã kết thúc */
function checkWindow_(exam, isResume) {
  var w = QC_window(exam.settings || {}), now = Date.now();
  if (w.from && now < w.from) throw new Error('Chưa đến giờ vào làm bài. Đề mở lúc ' + fmtVN_(w.from) + '.');
  if (w.end && now >= w.end) throw new Error('Bài kiểm tra đã kết thúc lúc ' + fmtVN_(w.end) + '.');
  if (!isResume && w.to && now > w.to) throw new Error('Đã hết giờ vào làm bài (chỉ cho vào đến ' + fmtVN_(w.to) + ').');
}
function listExams_() {
  var sh = sheet_(SH_EXAM, EXAM_HEAD);
  var res = rows_(sheet_(SH_RES, RES_HEAD));
  var cnt = {};
  res.forEach(function (r) { cnt[r[1]] = (cnt[r[1]] || 0) + 1; });
  return rows_(sh).filter(function (r) { return r[0]; }).map(function (r) {
    return { id: r[0], title: r[1], loai: r[2], mon: r[3], lop: r[4], status: r[5], created: r[6], updated: r[7], submitted: cnt[r[0]] || 0 };
  }).reverse();
}
function setStatus_(id, status) {
  var sh = sheet_(SH_EXAM, EXAM_HEAD);
  var row = findRow_(sh, 1, id);
  if (row < 0) throw new Error('Không tìm thấy đề');
  sh.getRange(row, 6).setValue(status === 'Mở' ? 'Mở' : 'Đóng');
  return { status: sh.getRange(row, 6).getValue() };
}
function deleteExam_(id) {
  return withLock_(function () {
    var sh = sheet_(SH_EXAM, EXAM_HEAD);
    var row = findRow_(sh, 1, id);
    if (row > 0) sh.deleteRow(row);
    return { deleted: row > 0 };
  });
}

/* ------------------------------- Lượt thi -------------------------------- */
function attKey_(hoTen, lop, mssv) { return QC_normName(hoTen) + '|' + QC_normName(lop) + '|' + QC_normName(mssv); }

function start_(req) {
  var exam = openExam_(req.examId);
  var st = exam.settings || {};
  var clean = function (v, n) { return String(v || '').replace(/\s+/g, ' ').trim().replace(/^[=+\-@]+/, '').slice(0, n); };
  var hoTen = clean(req.hoTen, 80), lop = clean(req.lop, 40), mssv = clean(req.mssv, 30);
  if (hoTen.length < 3) throw new Error('Vui lòng nhập đầy đủ Họ và tên');
  if (!lop) throw new Error('Vui lòng nhập Lớp');
  if (st.requireMssv && !mssv) throw new Error('Vui lòng nhập Mã số HSSV');
  if (st.requireApp && req.client !== 'app') throw new Error('Bài thi này chỉ làm được bằng phần mềm Trộn đề thi Online trên máy tính của phòng máy (không làm bằng điện thoại / trình duyệt).');
  if (st.roster && st.roster.length) {
    var nm = QC_normName(hoTen);
    var okR = st.roster.some(function (r) { return QC_normName(r).indexOf(nm) >= 0 || (mssv && QC_normName(r).indexOf(QC_normName(mssv)) >= 0); });
    if (!okR) throw new Error('Họ tên "' + hoTen + '" không có trong danh sách lớp của đề này. Kiểm tra lại chính tả hoặc báo giáo viên.');
  }
  var key = attKey_(hoTen, lop, mssv);
  return withLock_(function () {
    var sh = sheet_(SH_ATT, ATT_HEAD);
    var data = rows_(sh);
    var resume = null, done = 0, maxA = QC_maxAttempts(st);
    for (var i = 0; i < data.length; i++) {
      if (data[i][1] !== exam.id || data[i][9] !== key) continue;
      if (data[i][7] === 'Đã nộp') done++;
      else resume = { row: i + 2, r: data[i] };
    }
    checkWindow_(exam, !!resume);
    if (!resume && maxA > 0 && done >= maxA) {
      throw new Error(maxA === 1 ? 'Em đã nộp bài này rồi. Nếu cần làm lại, hãy báo giáo viên.' : 'Em đã làm đủ ' + maxA + ' lần cho phép của bài này.');
    }
    var attemptNo = resume ? (+resume.r[14] || done + 1) : done + 1;
    var now = new Date();
    var attemptId, seed, startedAt, idx = 0, qStart = null, saved = null, viol = 0;
    if (resume) {
      attemptId = resume.r[0]; seed = resume.r[2]; startedAt = new Date(resume.r[6]);
      idx = +resume.r[10] || 0; qStart = resume.r[11] ? new Date(resume.r[11]).getTime() : null;
      try { saved = resume.r[12] ? JSON.parse(resume.r[12]) : null; } catch (e) { saved = null; }
      viol = +resume.r[13] || 0;
    } else {
      attemptId = exam.id + '-' + newId_(8); seed = attemptId + '|' + Math.random(); startedAt = now;
      sh.appendRow([attemptId, exam.id, seed, hoTen, lop, mssv, now, 'Đang làm', '', key, 0, now, '', 0, attemptNo]);
      qStart = now.getTime();
    }
    var paper = QC_buildPaper(exam, seed);
    return {
      attemptId: attemptId, startedAt: startedAt.getTime(), serverNow: now.getTime(),
      resume: !!resume, idx: idx, qStart: qStart, saved: saved, violations: viol, attemptNo: attemptNo, maxAttempts: maxA,
      info: QC_publicInfo(exam), items: QC_publicPaper(exam, paper), book: QC_bookInfo(exam, paper)
    };
  });
}

function progress_(req) {
  var sh = sheet_(SH_ATT, ATT_HEAD);
  var row = findRow_(sh, 1, req.attemptId);
  if (row < 0) throw new Error('Không tìm thấy lượt làm bài');
  var r = sh.getRange(row, 1, 1, ATT_HEAD.length).getValues()[0];
  if (r[7] === 'Đã nộp') return { submitted: true };
  var now = new Date();
  var changedQ = +r[10] !== +req.idx;
  sh.getRange(row, 11, 1, 4).setValues([[+req.idx || 0, changedQ ? now : r[11], JSON.stringify(req.answers || []).slice(0, 45000), +req.violations || 0]]);
  return { serverNow: now.getTime() };
}

function submit_(req) {
  return withLock_(function () {
    var sh = sheet_(SH_ATT, ATT_HEAD);
    var row = findRow_(sh, 1, req.attemptId);
    if (row < 0) throw new Error('Không tìm thấy lượt làm bài');
    var r = sh.getRange(row, 1, 1, ATT_HEAD.length).getValues()[0];
    var exam = loadExam_(r[1]);
    var st = exam.settings || {};
    var resSh = sheet_(SH_RES, RES_HEAD);
    if (r[7] === 'Đã nộp') {
      var rr = findRow_(resSh, 1, req.attemptId);
      var old = rr > 0 ? resSh.getRange(rr, 6).getValue() : null;
      return { already: true, score: st.showScore !== false ? old : null };
    }
    var paper = QC_buildPaper(exam, r[2]);
    var g = QC_grade(exam, paper, req.answers || [], {});
    var now = new Date();
    var startedAt = new Date(r[6]);
    var secs = Math.round((now - startedAt) / 1000);
    var limit = QC_totalLimit(st, paper), unlimited = !limit;
    var notes = [];
    if (!unlimited && secs > limit + 180) notes.push('Nộp trễ ' + (secs - limit) + ' giây');
    var w = QC_window(st);
    if (w.end && now.getTime() > w.end + 120000) notes.push('Nộp sau giờ kết thúc ' + Math.round((now.getTime() - w.end) / 60000) + ' phút');
    var attemptNo = +r[14] || 1;
    if (req.reason) notes.push(req.reason);
    var viol = +req.violations || 0;
    var detail = JSON.stringify({ seed: r[2], answers: req.answers || [], d: g.details.map(function (d) { return d.type === 'mc' ? [d.b, d.q, d.chosen, d.ok ? 1 : 0] : d.type === 'fill' ? [d.b, d.q, d.given, d.oks.map(function (x) { return x ? 1 : 0; })] : d.type === 'speak' ? [d.b, d.q, { rec: d.rec, dur: d.dur }] : [d.b, d.q, d.text]; }), log: (req.log || []).slice(0, 200) });
    resSh.appendRow([req.attemptId, r[1], r[3], r[4], r[5], g.score, g.mcRight, g.mcCount, g.mcEarned, g.essayEarned, g.max,
      g.essayCount ? (g.essayPending ? 'Chưa chấm ' + g.essayPending + ' câu' : 'Đã chấm') : '', startedAt, now, secs, viol, notes.join('; '), detail.slice(0, 49000), '{}', attemptNo, g.skills ? JSON.stringify({ s: g.skills, t: g.toeic || null }) : '']);
    sh.getRange(row, 8, 1, 2).setValues([['Đã nộp', now]]);
    sh.getRange(row, 14).setValue(viol);
    var outp = { score: null, submittedAt: now.getTime() };
    if (st.showScore !== false) { outp.score = g.score; outp.right = g.mcRight; outp.mcCount = g.mcCount; outp.essayPending = g.essayPending; outp.scale = +st.scale || 10; }
    outp.attemptNo = attemptNo; outp.maxAttempts = QC_maxAttempts(st);
    if (st.showScore !== false && g.skills) { outp.skills = g.skills; outp.toeic = g.toeic || null; }
    if (st.showReview) outp.review = QC_review(exam, paper, g);
    return outp;
  });
}

/* ------------------------------- Kết quả -------------------------------- */
function results_(examId) {
  var res = rows_(sheet_(SH_RES, RES_HEAD)).filter(function (r) { return r[1] === examId; }).map(function (r) {
    var o = {}; RES_HEAD.forEach(function (h, i) { o[h] = r[i] instanceof Date ? r[i].getTime() : r[i]; }); return o;
  });
  var att = rows_(sheet_(SH_ATT, ATT_HEAD)).filter(function (r) { return r[1] === examId && r[7] !== 'Đã nộp'; }).map(function (r) {
    return { attemptId: r[0], hoTen: r[3], lop: r[4], mssv: r[5], startedAt: r[6] instanceof Date ? r[6].getTime() : r[6], idx: r[10], violations: r[13], lan: r[14] };
  });
  return { results: res, inProgress: att, serverNow: Date.now() };
}

function gradeEssay_(attemptId, scores) {
  return withLock_(function () {
    var resSh = sheet_(SH_RES, RES_HEAD);
    var rr = findRow_(resSh, 1, attemptId);
    if (rr < 0) throw new Error('Không tìm thấy bài làm');
    var r = resSh.getRange(rr, 1, 1, RES_HEAD.length).getValues()[0];
    var exam = loadExam_(r[1]);
    var det = JSON.parse(r[17]);
    var paper = QC_buildPaper(exam, det.seed);
    var g = QC_grade(exam, paper, det.answers, scores || {});
    resSh.getRange(rr, 6, 1, 7).setValues([[g.score, g.mcRight, g.mcCount, g.mcEarned, g.essayEarned, g.max, g.essayPending ? 'Chưa chấm ' + g.essayPending + ' câu' : 'Đã chấm']]);
    resSh.getRange(rr, 19).setValue(JSON.stringify(scores || {}));
    if (g.skills) resSh.getRange(rr, 21).setValue(JSON.stringify({ s: g.skills, t: g.toeic || null }));
    return { score: g.score };
  });
}

function deleteAttempt_(attemptId) {
  return withLock_(function () {
    var a = sheet_(SH_ATT, ATT_HEAD), r = sheet_(SH_RES, RES_HEAD);
    var x = findRow_(a, 1, attemptId); if (x > 0) a.deleteRow(x);
    var y = findRow_(r, 1, attemptId); if (y > 0) r.deleteRow(y);
    return { deleted: x > 0 || y > 0 };
  });
}

/* ----------------------- Tệp âm thanh trên Google Drive ----------------------- */
var CHUNK_BYTES = 3 * 1024 * 1024;
function folder_(name) { var it = DriveApp.getFoldersByName(name); return it.hasNext() ? it.next() : DriveApp.createFolder(name); }
function chunk_(file, part) {
  var blob = file.getBlob(), bytes = blob.getBytes();
  var parts = Math.max(1, Math.ceil(bytes.length / CHUNK_BYTES));
  part = Math.max(0, Math.min(parts - 1, +part || 0));
  return { b64: Utilities.base64Encode(bytes.slice(part * CHUNK_BYTES, (part + 1) * CHUNK_BYTES)), parts: parts, part: part, mime: blob.getContentType(), size: bytes.length };
}
/** Giáo viên tải file nghe lên => lưu vào thư mục TronDeThi_FileNghe, chia sẻ "Bất kỳ ai có đường liên kết" (chỉ xem) */
function uploadAudio_(req) {
  if (!req.b64) throw new Error('Thiếu dữ liệu file');
  var blob = Utilities.newBlob(Utilities.base64Decode(req.b64), req.mime || 'audio/mpeg', String(req.name || 'audio.mp3').slice(0, 120));
  var f = folder_(req.kind === 'page' ? 'TronDeThi_SachDe' : 'TronDeThi_FileNghe').createFile(blob);
  var shared = true;
  try { f.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) { shared = false; }
  return { id: f.getId(), name: f.getName(), size: f.getSize(), shared: shared };
}
/** Học sinh tải file nghe qua máy chủ (dự phòng khi không phát được trực tiếp từ Drive) */
function audio_(req) {
  var exam = loadExam_(req.examId);
  if (!QC_audioIds(exam)[req.id]) throw new Error('File nghe không thuộc đề này');
  return chunk_(DriveApp.getFileById(req.id), req.part);
}
/** Học sinh nộp bản ghi âm câu NÓI (lưu riêng tư trong Drive của giáo viên) */
function uploadSpeech_(req) {
  var sh = sheet_(SH_ATT, ATT_HEAD);
  var row = findRow_(sh, 1, req.attemptId);
  if (row < 0) throw new Error('Không tìm thấy lượt làm bài');
  var r = sh.getRange(row, 1, 1, ATT_HEAD.length).getValues()[0];
  if (r[7] === 'Đã nộp') throw new Error('Bài đã nộp, không ghi âm thêm được');
  if (!req.b64 || req.b64.length > 15000000) throw new Error('Bản ghi âm trống hoặc quá dài');
  var ext = /mp4|m4a|aac/.test(req.mime || '') ? '.m4a' : /ogg/.test(req.mime || '') ? '.ogg' : '.webm';
  var name = r[1] + '_' + QC_normName(r[3]).replace(/\s+/g, '_') + '_cau' + ((+req.idx || 0) + 1) + '_' + Date.now() + ext;
  var f = folder_('TronDeThi_GhiAm').createFile(Utilities.newBlob(Utilities.base64Decode(req.b64), req.mime || 'audio/webm', name));
  return { id: f.getId() };
}

/* =====================================================================
 * core.js — Lõi trộn đề / chấm điểm (dùng chung cho trình duyệt và
 * máy chủ Google Apps Script). Phần mềm: Lê Duy Nam — GV Trường CĐ Dầu khí

var QC_VERSION = '2.0';

function QC_hash(str) {
  var h1 = 0xdeadbeef ^ 0, h2 = 0x41c6ce57 ^ 0;
  str = String(str);
  for (var i = 0; i < str.length; i++) {
    var ch = str.charCodeAt(i);
    h1 = Math.imul(h1 ^ ch, 2654435761);
    h2 = Math.imul(h2 ^ ch, 1597334677);
  }
  h1 = Math.imul(h1 ^ (h1 >>> 16), 2246822507) ^ Math.imul(h2 ^ (h2 >>> 13), 3266489909);
  return h1 >>> 0;
}
function QC_rng(seedStr) {
  var a = QC_hash(seedStr);
  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    var t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
function QC_shuffle(arr, rnd) {
  for (var i = arr.length - 1; i > 0; i--) {
    var j = Math.floor(rnd() * (i + 1));
    var t = arr[i]; arr[i] = arr[j]; arr[j] = t;
  }
  return arr;
}
/** "1-10, 15, 20-22" -> [0..9,14,19,20,21] (chỉ số 0-based, trong phạm vi max) */
function QC_parseList(s, max) {
  var out = [], seen = {};
  String(s || '').split(/[,;\s]+/).forEach(function (part) {
    if (!part) return;
    var m = part.match(/^(\d+)\s*[-–]\s*(\d+)$/);
    var a, b;
    if (m) { a = +m[1]; b = +m[2]; } else if (/^\d+$/.test(part)) { a = b = +part; } else return;
    if (a > b) { var t = a; a = b; b = t; }
    for (var k = a; k <= b; k++) if (k >= 1 && k <= max && !seen[k]) { seen[k] = 1; out.push(k - 1); }
  });
  return out;
}
/** "30" -> 30 ; "25-30" -> số ngẫu nhiên trong khoảng */
function QC_parseCount(s, rnd) {
  var m = String(s == null ? '' : s).match(/^\s*(\d+)\s*(?:[-–]\s*(\d+))?\s*$/);
  if (!m) return 0;
  var a = +m[1], b = m[2] ? +m[2] : a;
  if (a > b) { var t = a; a = b; b = t; }
  return a + Math.floor(rnd() * (b - a + 1));
}
function QC_round(x, step) {
  step = +step || 0.1;
  var r = Math.round(x / step) * step;
  return Math.round(r * 100) / 100;
}
function QC_normName(s) {
  return String(s || '').normalize('NFD').replace(/[̀-ͯ]/g, '').replace(/đ/g, 'd').replace(/Đ/g, 'D')
    .replace(/\s+/g, ' ').trim().toUpperCase();
}
function QC_usable(q) {
  if (!q || q.disabled) return false;
  if (q.type === 'mc') return q.options && q.options.length >= 2 && q.correct && q.correct.length > 0;
  if (q.type === 'fill') return !!(q.blanks && q.blanks.length && q.blanks.every(function (b) { return b && b.length; }));
  return q.type === 'essay' || q.type === 'speak';
}
/** Chuẩn hoá câu trả lời điền khuyết: không phân biệt hoa/thường, dấu nháy cong/thẳng, khoảng trắng thừa, dấu câu cuối */
function QC_normAns(s) {
  return String(s == null ? '' : s).normalize('NFC').replace(/[\u2018\u2019\u02BC`´]/g, "'").replace(/[\u201C\u201D]/g, '"')
    .replace(/\s+/g, ' ').trim().replace(/^["'(]+|["'.!?,;:)]+$/g, '').trim().toLowerCase();
}

/**
 * Tạo đề cho 1 lượt thi.
 * exam.groups: [{bank, pick:'random'|'fixed', range, count, list, sec, pts, shuffleOpts, skill}]
 * exam.settings: {order:'mix'|'group'|'keep', shuffleOpts, sec, sameForAll}
 * Câu thuộc cùng một đoạn văn / đoạn nghe (q.g) luôn đi cùng nhau, đúng thứ tự (một "khối").
 * Đề không có đoạn dùng chung => kết quả trộn y hệt các phiên bản trước.
 */
function QC_buildPaper(exam, seedStr) {
  var st = exam.settings || {};
  var rnd = QC_rng(st.sameForAll ? ('same|' + exam.id) : seedStr);
  var used = {};
  var blocks = [], anyBook = false;
  (exam.groups || []).forEach(function (g, gi) {
    var bank = exam.banks[g.bank];
    if (!bank) return;
    // Đề TOEIC nguyên bộ (sách đề PDF + phiếu trả lời): giữ nguyên số câu, thứ tự và vị trí A/B/C/D
    var isBook = !!bank.book;
    if (isBook) anyBook = true;
    var qs = bank.questions || [];
    used[g.bank] = used[g.bank] || {};
    var okSkill = function (i) { return !g.skill || g.skill === 'all' || qs[i].skill === g.skill; };
    var toUnits = function (list) {
      var units = [], byKey = {};
      list.forEach(function (i) {
        var gk = qs[i].g;
        if (gk == null) { units.push([i]); return; }
        if (!byKey[gk]) { byKey[gk] = []; units.push(byKey[gk]); }
        byKey[gk].push(i);
      });
      units.forEach(function (u) { if (u.length > 1) u.sort(function (a, b) { return a - b; }); });
      return units;
    };
    var units;
    if (g.pick === 'fixed') {
      units = toUnits(QC_parseList(g.list, qs.length).filter(function (i) { return QC_usable(qs[i]) && !used[g.bank][i] && okSkill(i); }));
    } else if (isBook) {
      var bpool = (g.range && String(g.range).trim()) ? QC_parseList(g.range, qs.length) : qs.map(function (_, i) { return i; });
      units = bpool.filter(function (i) { return QC_usable(qs[i]) && !used[g.bank][i] && okSkill(i); }).sort(function (a, b) { return a - b; }).map(function (i) { return [i]; });
    } else {
      var pool = (g.range && String(g.range).trim()) ? QC_parseList(g.range, qs.length) : qs.map(function (_, i) { return i; });
      pool = pool.filter(function (i) { return QC_usable(qs[i]) && !used[g.bank][i] && okSkill(i); });
      var all = toUnits(pool);
      var n = QC_parseCount(g.count, rnd);
      if (!(n > 0) || n > pool.length) n = pool.length;
      var sh = QC_shuffle(all.slice(), rnd);
      units = []; var total = 0;
      for (var k = 0; k < sh.length && total < n; k++) { if (total + sh[k].length <= n) { units.push(sh[k]); total += sh[k].length; } }
      if (st.order === 'keep') units.sort(function (a, b) { return a[0] - b[0]; });
    }
    var unitItems = units.map(function (u) {
      return u.map(function (qi) {
        used[g.bank][qi] = 1;
        var q = qs[qi];
        var perm = null;
        if (q.type === 'mc') {
          perm = q.options.map(function (_, i) { return i; });
          // câu NGHE không trộn phương án (băng đọc A, B, C, D theo thứ tự)
          var doShuffle = !isBook && st.shuffleOpts !== false && g.shuffleOpts !== false && q.skill !== 'L';
          if (doShuffle) {
            var pin = q.pin || [];
            var free = perm.filter(function (i) { return !pin[i]; });
            var shp = QC_shuffle(free.slice(), rnd);
            var kk = 0;
            perm = perm.map(function (i) { return pin[i] ? i : shp[kk++]; });
          }
        }
        var sec = (q.sec != null && q.sec !== '') ? +q.sec : (g.sec != null && g.sec !== '' ? +g.sec : (st.sec != null && st.sec !== '' ? +st.sec : 60));
        var pts = (q.pts != null && q.pts !== '') ? +q.pts : (g.pts != null && g.pts !== '' ? +g.pts : 1);
        var it = { b: g.bank, q: qi, g: gi, perm: perm, sec: sec, pts: pts };
        if (q.g != null) it.grp = g.bank + ':' + q.g;
        return it;
      });
    });
    if (st.order === 'group' && !isBook) QC_shuffle(unitItems, rnd);
    blocks.push(unitItems);
  });
  var allUnits = [].concat.apply([], blocks);
  if ((st.order === 'mix' || !st.order) && !anyBook) QC_shuffle(allUnits, rnd);
  return { items: [].concat.apply([], allUnits) };
}

/** Link nghe: tên file đã tải lên (bank.audio) hoặc link Google Drive / mp3 */
function QC_audioRef(bank, ref) {
  if (!ref) return null;
  ref = String(ref).trim();
  if (/^https?:\/\//i.test(ref)) {
    var m = ref.match(/drive\.google\.com\/(?:file\/d\/|open\?(?:.*&)?id=|uc\?(?:.*&)?id=)([\w-]{20,})/);
    return m ? { id: m[1], src: 'https://drive.google.com/uc?export=download&id=' + m[1] } : { id: null, src: ref };
  }
  var a = bank && bank.audio && (bank.audio[ref] || bank.audio[ref.toLowerCase()]);
  if (a && a.id) return { id: a.id, src: 'https://drive.google.com/uc?export=download&id=' + a.id };
  if (a && a.link) return QC_audioRef(bank, a.link);
  return null;
}
/** Danh sách mã file Drive được phép phát cho đề này (chống đọc file Drive tùy ý) */
function QC_audioIds(exam) {
  var ids = {};
  (exam.banks || []).forEach(function (b) {
    Object.keys(b.audio || {}).forEach(function (k) { if (b.audio[k] && b.audio[k].id) ids[b.audio[k].id] = 1; });
    var chk = function (ref) { var r = QC_audioRef(b, ref); if (r && r.id) ids[r.id] = 1; };
    (b.questions || []).forEach(function (q) { chk(q.audio); });
    (b.groups || []).forEach(function (g) { chk(g && g.audio); });
    if (b.book) {
      (b.book.pages || []).forEach(function (pg) { if (pg && pg.id) ids[pg.id] = 1; });
      Object.keys(b.book.parts || {}).forEach(function (k) { chk(b.book.parts[k]); });
    }
  });
  return ids;
}
/** Ảnh trang sách đề trên Google Drive (chia sẻ "ai có link") */
function QC_pageSrc(id) { return 'https://lh3.googleusercontent.com/d/' + id + '=w1600'; }
/** Sách đề + file nghe theo Part của các ngân hàng "đề nguyên bộ" có trong đề này (gửi cho học sinh khi bắt đầu làm) */
function QC_bookInfo(exam, paper) {
  var out = null;
  paper.items.forEach(function (it) {
    var b = exam.banks[it.b];
    if (!b || !b.book || (out && out[it.b])) return;
    out = out || {};
    var parts = {};
    Object.keys(b.book.parts || {}).forEach(function (k) { var r = QC_audioRef(b, b.book.parts[k]); parts[k] = r || { missing: b.book.parts[k] }; });
    out[it.b] = {
      name: b.name || '',
      pages: (b.book.pages || []).map(function (pg) { return { id: pg.id || null, src: pg.id ? QC_pageSrc(pg.id) : (pg.src || ''), sec: pg.sec || '', w: pg.w || 0, h: pg.h || 0 }; }),
      parts: parts,
      marks: b.book.marks || null
    };
  });
  return out;
}

/** Đề gửi cho học sinh — KHÔNG có đáp án */
function QC_publicPaper(exam, paper) {
  var pos = {}, size = {};
  paper.items.forEach(function (it) { if (it.grp) size[it.grp] = (size[it.grp] || 0) + 1; });
  return paper.items.map(function (it) {
    var bank = exam.banks[it.b];
    var q = bank.questions[it.q];
    var o = { type: q.type, stem: q.stem, sec: it.sec, pts: it.pts };
    if (q.skill) o.skill = q.skill;
    if (bank.book) { o.bk = it.b; o.num = q.num; o.part = q.part; o.page = q.page == null ? null : q.page; }
    if (q.type === 'mc') {
      o.options = it.perm.map(function (i) { return q.options[i]; });
      o.multi = q.correct.length > 1;
    } else if (q.type === 'fill') {
      o.nb = q.blanks.length;
    } else if (q.type === 'speak') {
      o.rec = +q.rec || 60;
    }
    var au = QC_audioRef(bank, q.audio);
    if (q.audio) o.audio = au || { missing: q.audio };
    if (it.grp) {
      var gr = (bank.groups || [])[q.g] || {};
      pos[it.grp] = (pos[it.grp] || 0) + 1;
      o.group = { key: it.grp, title: gr.title || '', stem: gr.stem || '', pos: pos[it.grp], size: size[it.grp] };
      if (gr.audio) o.group.audio = QC_audioRef(bank, gr.audio) || { missing: gr.audio };
    }
    return o;
  });
}

/** answers[i]: mảng chỉ số phương án ĐANG HIỂN THỊ (trắc nghiệm) hoặc chuỗi (tự luận) */
function QC_grade(exam, paper, answers, essayScores) {
  var st = exam.settings || {};
  var scale = +st.scale || 10;
  var res = { mcRight: 0, mcCount: 0, essayCount: 0, earned: 0, max: 0, mcEarned: 0, essayEarned: 0, essayMax: 0, essayPending: 0, details: [] };
  essayScores = essayScores || {};
  paper.items.forEach(function (it, i) {
    var q = exam.banks[it.b].questions[it.q];
    var a = answers ? answers[i] : null;
    res.max += it.pts;
    if (q.type === 'mc') {
      res.mcCount++;
      var chosen = (Array.isArray(a) ? a : (a == null || a === '' ? [] : [a])).map(function (d) { return it.perm[+d]; }).filter(function (x) { return x != null; });
      chosen.sort(); var corr = q.correct.slice().sort();
      var ok = chosen.length === corr.length && chosen.every(function (x, k) { return x === corr[k]; });
      if (ok) { res.mcRight++; res.mcEarned += it.pts; }
      res.details.push({ type: 'mc', b: it.b, q: it.q, chosen: chosen, correct: corr, ok: ok, pts: it.pts, earned: ok ? it.pts : 0 });
    } else if (q.type === 'fill') {
      res.mcCount++;
      var given = Array.isArray(a) ? a : (a == null ? [] : [a]);
      var oks = q.blanks.map(function (alts, k) {
        var g = QC_normAns(given[k]);
        return !!g && alts.some(function (x) { return QC_normAns(x) === g; });
      });
      var nOk = oks.filter(Boolean).length;
      var earnedF = Math.round(it.pts * nOk / q.blanks.length * 100) / 100;
      if (nOk === q.blanks.length) res.mcRight++;
      res.mcEarned += earnedF;
      res.details.push({ type: 'fill', b: it.b, q: it.q, given: q.blanks.map(function (_, k) { return String(given[k] == null ? '' : given[k]).slice(0, 300); }), oks: oks, ok: nOk === q.blanks.length, pts: it.pts, earned: earnedF });
    } else {
      res.essayCount++;
      res.essayMax += it.pts;
      var sc = essayScores[i];
      var graded = sc != null && sc !== '';
      var e = graded ? Math.max(0, Math.min(it.pts, +sc)) : 0;
      if (!graded) res.essayPending++;
      res.essayEarned += e;
      if (q.type === 'speak') {
        var rec = (a && typeof a === 'object' && a.rec) ? String(a.rec) : '';
        res.details.push({ type: 'speak', b: it.b, q: it.q, rec: rec, dur: a && a.dur ? +a.dur : 0, pts: it.pts, earned: e, graded: graded });
      } else {
        res.details.push({ type: 'essay', b: it.b, q: it.q, text: String(a == null ? '' : a).slice(0, 20000), pts: it.pts, earned: e, graded: graded });
      }
    }
    if (q.skill) {
      var sk = res.skills || (res.skills = {});
      var o = sk[q.skill] || (sk[q.skill] = { earned: 0, max: 0, right: 0, count: 0, pending: 0 });
      var d = res.details[res.details.length - 1];
      o.max += it.pts; o.earned += d.earned; o.count++;
      if (d.ok) o.right++;
      if ((d.type === 'essay' || d.type === 'speak') && !d.graded) o.pending++;
    }
  });
  res.earned = res.mcEarned + res.essayEarned;
  res.score = res.max > 0 ? QC_round(res.earned / res.max * scale, st.round || 0.1) : 0;
  if (res.skills) {
    Object.keys(res.skills).forEach(function (k) { var o = res.skills[k]; o.earned = Math.round(o.earned * 100) / 100; o.score = o.max > 0 ? QC_round(o.earned / o.max * 10, 0.1) : 0; });
    if (st.toeic) {
      // Quy đổi TOEIC THAM KHẢO (tuyến tính): Nghe 5–495, Đọc 5–495, Viết / Nói 0–200
      var t = {}, lin = function (o, lo, hi, step) { return o && o.max ? Math.round((lo + (hi - lo) * o.earned / o.max) / step) * step : null; };
      t.L = lin(res.skills.L, 5, 495, 5); t.R = lin(res.skills.R, 5, 495, 5);
      t.W = lin(res.skills.W, 0, 200, 10); t.S = lin(res.skills.S, 0, 200, 10);
      if (t.L != null || t.R != null) t.LR = (t.L || 0) + (t.R || 0);
      res.toeic = t;
    }
  }
  return res;
}

/** Khung giờ (mốc thời gian ms): from = bắt đầu cho vào, to = hết giờ vào làm, end = kết thúc bài (tự nộp) */
function QC_window(st) {
  st = st || {};
  var ts = function (n) { var v = st[n + 'Ts']; if (v) return +v; if (st[n]) { var d = new Date(st[n]).getTime(); return isNaN(d) ? 0 : d; } return 0; };
  return { from: ts('openFrom'), to: ts('openTo'), end: ts('endAt') };
}
/** Số lần được làm: 0 = không giới hạn (luyện tập) */
function QC_maxAttempts(st) {
  st = st || {};
  if (st.unlimited) return 0;
  if (st.maxAttempts != null && st.maxAttempts !== '') return Math.max(0, Math.floor(+st.maxAttempts) || 0);
  return st.allowRetake ? 0 : 1;
}
/** Đáp án chi tiết trả về học sinh sau khi nộp (chỉ khi giáo viên bật "Cho xem đáp án") */
function QC_review(exam, paper, g) {
  return paper.items.map(function (it, i) {
    var q = exam.banks[it.b].questions[it.q], d = g.details[i];
    if (q.type === 'mc') {
      var disp = function (o) { return it.perm.indexOf(o); };
      return { type: 'mc', num: q.num, stem: q.stem, options: it.perm.map(function (o) { return q.options[o]; }), chosen: d.chosen.map(disp), correct: q.correct.map(disp), ok: d.ok };
    }
    if (q.type === 'fill') return { type: 'fill', stem: q.stem, given: d.given, oks: d.oks, blanks: q.blanks, ok: d.ok };
    if (q.type === 'speak') return { type: 'speak', stem: q.stem, rec: d.rec, answer: q.answer || '' };
    return { type: 'essay', stem: q.stem, text: d.text, answer: q.answer || '' };
  });
}

/** Thời gian tối đa cả bài (giây), 0 = không giới hạn.
 *  navMode 'strict': cộng thời gian từng câu; 'free': không giới hạn (ôn tập) hoặc tổng thời gian giáo viên đặt */
function QC_totalLimit(st, paper) {
  st = st || {};
  var sum = 0, unl = false;
  paper.items.forEach(function (it) { if (it.sec > 0) sum += it.sec; else unl = true; });
  if (st.navMode === 'free') {
    if (st.freeTime !== 'total') return 0;
    if (+st.totalMin > 0) return Math.round(+st.totalMin * 60);
  }
  return unl ? 0 : sum;
}

/** Thông tin đề công khai cho học sinh */
function QC_publicInfo(exam) {
  var st = exam.settings || {};
  var paper = QC_buildPaper(exam, 'preview');
  var totalSec = QC_totalLimit(st, paper);
  return {
    id: exam.id, title: exam.title || '', info: exam.info || {},
    count: paper.items.length, totalSec: totalSec, navMode: st.navMode === 'free' ? 'free' : 'strict',
    requireMssv: !!st.requireMssv, hasRoster: !!(st.roster && st.roster.length),
    maxViolations: +st.maxViolations || 0, fullscreen: st.fullscreen !== false,
    showScore: st.showScore !== false, scale: +st.scale || 10,
    window: QC_window(st), maxAttempts: QC_maxAttempts(st), showReview: !!st.showReview, requireApp: !!st.requireApp,
    kind: exam.kind === 'en' ? 'en' : 'subject', audioPlays: st.audioPlays != null && st.audioPlays !== '' ? +st.audioPlays : 1,
    bkAuto: st.bkAuto !== false, bkLockL: !!st.bkLockL, bkLockSec: !!st.bkLockSec,
    hasSpeak: paper.items.some(function (it) { return exam.banks[it.b].questions[it.q].type === 'speak'; }),
    hasAudio: paper.items.some(function (it) { var b = exam.banks[it.b], q = b.questions[it.q]; return !!(q.audio || (b.book && q.skill === 'L') || (q.g != null && b.groups && b.groups[q.g] && b.groups[q.g].audio)); }),
    booklet: paper.items.some(function (it) { return !!exam.banks[it.b].book; }),
    note: st.note || '',
    music: st.music && st.music.url ? { url: String(st.music.url), vol: +st.music.vol || 35, allowMute: st.music.allowMute !== false } : null
  };
}

var QuizCore = {
